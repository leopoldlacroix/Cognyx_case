def detect_component_identities(conn: sqlite3.Connection, config: Optional[Dict] = None) -> List[Dict]:
    """Detect identity/alias relationships between PLM components and ERP materials.

    Matches on normalized reference equality across source systems.
    """
    if config is None:
        config = load_normalization_config()

    matches = []

    # Get all PLM BOM components with normalized refs
    plm_components = conn.execute('''
        SELECT DISTINCT component_ref_normalized, 'PLM' as source_system,
               'plm_bom_line' as source_table, component_ref_raw as raw_id
        FROM plm_bom_line
        WHERE component_ref_normalized IS NOT NULL
        AND component_ref_normalized != ''
    ''').fetchall()

    # Get all ERP materials with normalized IDs
    erp_materials = conn.execute('''
        SELECT DISTINCT material_id_normalized, 'ERP' as source_system,
               'erp_material' as source_table, material_id_raw as raw_id
        FROM erp_material
        WHERE material_id_normalized IS NOT NULL
        AND material_id_normalized != ''
    ''').fetchall()

    # Build lookup: normalized_ref → list of source records
    plm_by_ref = {}
    for c in plm_components:
        ref = c['component_ref_normalized']
        if ref not in plm_by_ref:
            plm_by_ref[ref] = []
        plm_by_ref[ref].append(c)

    erp_by_ref = {}
    for m in erp_materials:
        ref = m['material_id_normalized']
        if ref not in erp_by_ref:
            erp_by_ref[ref] = []
        erp_by_ref[ref].append(m)

    # Find overlaps
    common_refs = set(plm_by_ref.keys()) & set(erp_by_ref.keys())
    for ref in sorted(common_refs):
        for plm in plm_by_ref[ref]:
            for erp in erp_by_ref[ref]:
                match = {
                    'canonical_ref': ref,
                    'source_system_a': plm['source_system'],
                    'source_table_a': plm['source_table'],
                    'raw_id_a': plm['raw_id'],
                    'source_system_b': erp['source_system'],
                    'source_table_b': erp['source_table'],
                    'raw_id_b': erp['raw_id'],
                    'match_type': 'identity',
                    'confidence': 'high',
                    'match_method': 'normalized_reference_exact',
                    'status': 'pending',
                }
                matches.append(match)
                record_reconciliation(conn, match)

    return matches
```

def detect_supplier_identities(conn: sqlite3.Connection, config: Optional[Dict] = None) -> List[Dict]:
    """Detect identity relationships between PLM BOM suppliers and ERP supplier master."""
    if config is None:
        config = load_normalization_config()

    matches = []
    supplier_aliases = config.get('supplier_aliases', {})

    # Get PLM BOM suppliers (normalized)
    plm_suppliers = conn.execute('''
        SELECT DISTINCT supplier_normalized, supplier_raw
        FROM plm_bom_line
        WHERE supplier_normalized IS NOT NULL
        AND supplier_normalized != ''
    ''').fetchall()

    # Get ERP suppliers
    erp_suppliers = conn.execute('''
        SELECT DISTINCT supplier_name_normalized, supplier_name_raw
        FROM erp_supplier
        WHERE supplier_name_normalized IS NOT NULL
        AND supplier_name_normalized != ''
    ''').fetchall()

    # Build lookup
    erp_by_name = {}
    for s in erp_suppliers:
        erp_by_name[s['supplier_name_normalized']] = s

    for plm_s in plm_suppliers:
        plm_norm = plm_s['supplier_normalized']
        # Try exact match first
        if plm_norm in erp_by_name:
            erp_s = erp_by_name[plm_norm]
            match = {
                'canonical_ref': plm_norm,
                'source_system_a': 'PLM',
                'source_table_a': 'plm_bom_line',
                'raw_id_a': plm_s['supplier_raw'],
                'source_system_b': 'ERP',
                'source_table_b': 'erp_supplier',
                'raw_id_b': erp_s['supplier_name_raw'],
                'match_type': 'identity',
                'confidence': 'high',
                'match_method': 'supplier_name_normalized_exact',
                'status': 'pending',
            }
            matches.append(match)
            record_reconciliation(conn, match)

    return matches
```

    report['suppliers'] = len(sup_matches)
    print(f"  Found {len(sup_matches)} supplier identity matches")

    report['total'] = len(comp_matches) + len(sup_matches)
    return report
