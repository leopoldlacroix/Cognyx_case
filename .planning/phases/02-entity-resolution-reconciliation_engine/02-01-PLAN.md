- When supplier_id matches a known supplier, supplier_name_normalized is populated from erp_supplier.supplier_name_normalized
- When supplier_id doesn't match any known supplier, supplier_name_normalized remains NULL and a soft warning is emitted
- The mapping is done during a normalization pass, not during initial ingestion
</acceptance_criteria>

<action>
Create `normalize_erp_materials()` function in `app/services/normalization.py`:

```python
def normalize_erp_materials(conn: sqlite3.Connection) -> Dict[str, Any]:
    """Normalize ERP material supplier references using supplier master lookup.

    For each erp_material row:
    - Look up supplier_id_raw in erp_supplier.supplier_id_normalized
    - If found, populate supplier_name_normalized from supplier master
    - If not found, leave NULL and emit warning
    """
    stats = {'processed': 0, 'matched': 0, 'unmatched': 0}

    # Build supplier lookup: normalized_id → normalized_name
    supplier_lookup = {}
    for row in conn.execute('SELECT supplier_id_normalized, supplier_name_normalized FROM erp_supplier WHERE supplier_id_normalized IS NOT NULL AND supplier_name_normalized IS NOT NULL').fetchall():
        supplier_lookup[row['supplier_id_normalized']] = row['supplier_name_normalized']

    # Update ERP materials
    for row in conn.execute('SELECT id, supplier_id_raw FROM erp_material WHERE supplier_id_raw IS NOT NULL').fetchall():
        stats['processed'] += 1
        # Try to match against normalized supplier IDs
        matched_name = None
        for sup_id_norm, sup_name_norm in supplier_lookup.items():
            if row['supplier_id_raw'].strip().upper() == sup_id_norm:
                matched_name = sup_name_norm
                break

        if matched_name:
            conn.execute('UPDATE erp_material SET supplier_name_normalized = ? WHERE id = ?',
                        (matched_name, row['id']))
            stats['matched'] += 1
        else:
            stats['unmatched'] += 1

    conn.commit()
    return stats
```

```python
def normalize_erp_materials(conn: sqlite3.Connection, config: Optional[Dict] = None) -> Dict[str, Any]:
    """Normalize ERP material supplier references and UOM values."""
    if config is None:
        config = load_normalization_config()

    stats = {'processed': 0, 'matched': 0, 'unmatched': 0, 'uom_normalized': 0, 'uom_unknown': 0}

    # Build supplier lookup
    supplier_lookup = {}
    for row in conn.execute('SELECT supplier_id_normalized, supplier_name_normalized FROM erp_supplier WHERE supplier_id_normalized IS NOT NULL AND supplier_name_normalized IS NOT NULL').fetchall():
        supplier_lookup[row['supplier_id_normalized']] = row['supplier_name_normalized']

    uom_aliases = config.get('uom_aliases', {})

    for row in conn.execute('SELECT id, supplier_id_raw, base_unit_raw FROM erp_material WHERE supplier_id_raw IS NOT NULL').fetchall():
        stats['processed'] += 1

        # Supplier mapping
        matched_name = None
        for sup_id_norm, sup_name_norm in supplier_lookup.items():
            if row['supplier_id_raw'].strip().upper() == sup_id_norm:
                matched_name = sup_name_norm
                break

        if matched_name:
            conn.execute('UPDATE erp_material SET supplier_name_normalized = ? WHERE id = ?',
                        (matched_name, row['id']))
            stats['matched'] += 1

        # UOM standardization
        if row['base_unit_raw']:
            raw_uom = row['base_unit_raw'].strip().upper()
            normalized_uom = uom_aliases.get(raw_uom, raw_uom)
            conn.execute('UPDATE erp_material SET base_unit_normalized = ? WHERE id = ?',
                        (normalized_uom, row['id']))
            stats['uom_normalized'] += 1

    conn.commit()
    return stats
```
