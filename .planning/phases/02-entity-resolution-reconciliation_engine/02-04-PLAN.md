```python
def detect_functional_similarity(conn: sqlite3.Connection, config: Optional[Dict] = None) -> List[Dict]:
    """Detect functionally similar but non-identical components.

    Similarity criteria:
    - Same reference prefix (e.g., DOOR-CTRL) but different suffix
    - Same component type but different variant qualifier
    NOT identity: different normalized references that are not aliases
    """
    if config is None:
        config = load_normalization_config()

    candidates = []

    # Get all distinct normalized component references
    refs = conn.execute('''
        SELECT DISTINCT component_ref_normalized
        FROM plm_bom_line
        WHERE component_ref_normalized IS NOT NULL
        AND component_ref_normalized != ''
        ORDER BY component_ref_normalized
    ''').fetchall()

    ref_list = [r['component_ref_normalized'] for r in refs]

    # Find similar pairs: same prefix, different suffix
    for i, ref_a in enumerate(ref_list):
        for ref_b in ref_list[i+1:]:
            # Skip if one is alias of the other
            if ref_a == ref_b:
                continue

            # Check prefix similarity
            parts_a = ref_a.split('-')
            parts_b = ref_b.split('-')

            if len(parts_a) >= 2 and len(parts_b) >= 2:
                # Same prefix (first 2 parts), different suffix
                prefix_a = '-'.join(parts_a[:2])
                prefix_b = '-'.join(parts_b[:2])

                if prefix_a == prefix_b and parts_a[-1] != parts_b[-1]:
                    candidate = {
                        'reference_a': ref_a,
                        'reference_b': ref_b,
                        'similarity_type': 'same_prefix_different_suffix',
                        'similarity_detail': f"Prefix '{prefix_a}' matches, suffixes differ: '{parts_a[-1]}' vs '{parts_b[-1]}'",
                        'match_type': 'functional_similarity',
                        'confidence': 'medium',
                        'match_method': 'prefix_similarity',
                        'status': 'pending_review',
                        'rationale': f"Similar component family but different variants — requires human review to determine if interchangeable",
                    }
                    candidates.append(candidate)

    return candidates
```

```python
def detect_variant_specific_differences(conn: sqlite3.Connection) -> List[Dict]:
    """Detect components used only by specific variants (variant-specific specialization).

    Nordic climate variants (REGIO-NORDIC, INTEGRA-NORDIC) use cold-rated parts
    that should NOT be flagged as reconciliation candidates with standard parts.
    """
    # Get Nordic variant refs
    nordic_variants = conn.execute('''
        SELECT variant_ref_normalized
        FROM plm_variant
        WHERE variant_name_raw LIKE '%nordic%' OR variant_name_raw LIKE '%Nordic%'
        AND variant_ref_normalized IS NOT NULL
    ''').fetchall()

    nordic_refs = {v['variant_ref_normalized'] for v in nordic_variants}

    if not nordic_refs:
        return []

    # Find BOM lines used only by Nordic variants
    nordic_only_components = conn.execute('''
        SELECT DISTINCT component_ref_normalized
        FROM plm_bom_line
        WHERE variant_ref_normalized IN (SELECT variant_ref_normalized FROM plm_variant WHERE variant_name_raw LIKE '%nordic%' OR variant_name_raw LIKE '%Nordic%')
        AND component_ref_normalized NOT IN (
            SELECT component_ref_normalized
            FROM plm_bom_line
            WHERE variant_ref_normalized NOT IN (SELECT variant_ref_normalized FROM plm_variant WHERE variant_name_raw LIKE '%nordic%' OR variant_name_raw LIKE '%Nordic%')
        )
    ''').fetchall()

    records = []
    for comp in nordic_only_components:
        record = {
            'component_ref': comp['component_ref_normalized'],
            'match_type': 'variant_specific',
            'confidence': 'high',
            'match_method': 'variant_exclusive_usage',
            'status': 'excluded',
            'variant_ref': list(nordic_refs)[0],
            'rationale': 'Component used exclusively by Nordic climate variant — intentional specialization, not a reconciliation candidate',
        }
        records.append(record)

    return records
```

```python
def record_reconciliation(conn: sqlite3.Connection,
                          source_system_a: str, source_table_a: str, raw_id_a: str,
                          source_system_b: str, source_table_b: str, raw_id_b: str,
                          canonical_ref: Optional[str] = None,
                          match_type: str = 'identity',
                          match_method: str = 'unknown',
                          confidence: str = 'low',
                          status: str = 'pending',
                          rationale: Optional[str] = None,
                          evidence: Optional[Dict] = None) -> int:
    """Record a reconciliation proposal. Returns the reconciliation ID."""
    evidence_json = json.dumps(evidence) if evidence else None

    conn.execute('''
        INSERT INTO reconciliation (
            source_system_a, source_table_a, raw_id_a,
            source_system_b, source_table_b, raw_id_b,
            canonical_ref, match_type, match_method, confidence, status,
            rationale, evidence, created_at, updated_at
        ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
    ''', (
        source_system_a, source_table_a, raw_id_a,
        source_system_b, source_table_b, raw_id_b,
        canonical_ref, match_type, match_method, confidence, status,
        rationale, evidence_json,
        datetime.now(timezone.utc).isoformat(),
        datetime.now(timezone.utc).isoformat(),
    ))
    conn.commit()
    return conn.execute('SELECT last_insert_rowid()').fetchone()[0]
```

```python
def run_full_reconciliation(conn: sqlite3.Connection, config: Optional[Dict] = None) -> Dict:
    """Run complete reconciliation pipeline."""
    if config is None:
        config = load_normalization_config()

    report = {
        'component_identities': 0,
        'supplier_identities': 0,
        'functional_similarity': 0,
        'variant_specific': 0,
        'total': 0,
    }

    print("\n=== Running Reconciliation Pipeline ===")

    print("\n1. Component identity detection...")
    comp_ids = detect_component_identities(conn, config)
    report['component_identities'] = len(comp_ids)
    print(f"   Found {len(comp_ids)} identity matches")

    print("2. Supplier identity detection...")
    sup_ids = detect_supplier_identities(conn, config)
    report['supplier_identities'] = len(sup_ids)
    print(f"   Found {len(sup_ids)} supplier matches")

    print("3. Functional similarity detection...")
    func_sim = detect_functional_similarity(conn, config)
    report['functional_similarity'] = len(func_sim)
    print(f"   Found {len(func_sim)} similarity candidates")

    print("4. Variant-specific detection...")
    var_spec = detect_variant_specific_differences(conn)
    report['variant_specific'] = len(var_spec)
    print(f"   Found {len(var_spec)} variant-specific components")

    report['total'] = sum(report.values())
    print(f"\n=== Reconciliation Summary ===")
    print(f"Total records: {report['total']}")
    for key, value in report.items():
        if key != 'total':
            print(f"  {key}: {value}")

    return report
```
