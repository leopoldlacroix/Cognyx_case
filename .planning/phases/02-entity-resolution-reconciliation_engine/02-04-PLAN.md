# PLAN-2.4: Entity Resolution — Identity & Alias Detection

**Phase:** 2 — Entity Resolution & Reconciliation Engine  \
**Goal:** Identity and alias relationships between source entities are detected with confidence scoring and match method recording.

**Requirements addressed:** RECON-01, SCEN-B

**Depends on:** 02-03 (Schema Refinement + Source Entity Extraction) — requires source_component and source_supplier tables populated

**D-2.3 (mandatory):** `detect_component_identities()` and `detect_supplier_identities()` query **`source_component` / `source_supplier` only**. Do not scan raw `plm_bom_line` / `erp_material` / `erp_supplier` for identity matching — those are ingestion tables; source entities are populated by Plan 02-03 extraction.

**Deliverables:**
- Identity/alias detection across PLM source components, ERP source components, PLM/ERP source suppliers
- Confidence scoring for each match (REAL 0–1)
- Method recording using canonical values (EXACT / NORMALIZED / …)
- Reconciliation records created for each detected relationship via `record_reconciliation()` (schema from 02-03)

```python
def detect_component_identities(conn: sqlite3.Connection, config: Optional[Dict] = None) -> List[Dict]:
    """Detect identity/alias relationships between PLM source components and ERP source components.

    Matches on normalized_reference equality across source systems.
    Uses source_component table per D-2.3 (not raw tables).
    """
    if config is None:
        config = load_normalization_config()

    matches = []
    # Get PLM source components
    plm_components = conn.execute('''
        SELECT id, normalized_reference, source_reference, source_system,
               'source_component' as source_table
        FROM source_component
        WHERE source_system = 'PLM'
        AND normalized_reference IS NOT NULL
        AND normalized_reference != ''
    ''').fetchall()

    # Get ERP source components
    erp_components = conn.execute('''
        SELECT id, normalized_reference, source_reference, source_system,
               'source_component' as source_table
        FROM source_component
        WHERE source_system = 'ERP'
        AND normalized_reference IS NOT NULL
        AND normalized_reference != ''
    ''').fetchall()

    # Build lookup: normalized_ref -> list of source components
    plm_by_ref = {}
    for c in plm_components:
        ref = c['normalized_reference']
        plm_by_ref.setdefault(ref, []).append(c)

    erp_by_ref = {}
    for m in erp_components:
        ref = m['normalized_reference']
        erp_by_ref.setdefault(ref, []).append(m)

    # Find overlaps
    common_refs = set(plm_by_ref.keys()) & set(erp_by_ref.keys())
    for ref in sorted(common_refs):
        for plm in plm_by_ref[ref]:
            for erp in erp_by_ref[ref]:
                match = {
                    'canonical_ref': ref,
                    'source_system_a': plm['source_system'],
                    'source_table_a': plm['source_table'],
                    'raw_id_a': plm['source_reference'],
                    'source_component_id_a': plm['id'],
                    'source_system_b': erp['source_system'],
                    'source_table_b': erp['source_table'],
                    'raw_id_b': erp['source_reference'],
                    'source_component_id_b': erp['id'],
                    'relationship': 'identity',
                    'confidence': 0.95,  # REAL
                    'method': 'NORMALIZED',  # canonical method enum
                    'status': 'PENDING',
                }
                matches.append(match)
                # Persist via canonical component_reconciliation (02-03); evidence_json holds pair
                record_reconciliation(
                    conn,
                    entity_type='component',
                    source_entity_id=plm['id'],
                    status='PENDING',
                    method='NORMALIZED',
                    confidence=0.95,
                    rationale=f"Identity with ERP source_component {erp['id']} on {ref}",
                    evidence=match,
                )

    return matches
```

```python
def detect_supplier_identities(conn: sqlite3.Connection, config: Optional[Dict] = None) -> List[Dict]:
    """Detect identity relationships between PLM and ERP source suppliers."""
    if config is None:
        config = load_normalization_config()

    matches = []
    supplier_aliases = config.get('supplier_aliases', {})

    # Get all source suppliers grouped by source system
    plm_suppliers = conn.execute('''
        SELECT id, normalized_reference, source_reference, source_system
        FROM source_supplier
        WHERE source_system = 'PLM'
        AND normalized_reference IS NOT NULL
        AND normalized_reference != ''
    ''').fetchall()

    erp_suppliers = conn.execute('''
        SELECT id, normalized_reference, source_reference, source_system
        FROM source_supplier
        WHERE source_system = 'ERP'
        AND normalized_reference IS NOT NULL
        AND normalized_reference != ''
    ''').fetchall()

    # Build lookup
    erp_by_name = {}
    for s in erp_suppliers:
        erp_by_name[s['normalized_reference']] = s

    for plm_s in plm_suppliers:
        plm_norm = plm_s['normalized_reference']
        # Try exact match first
        if plm_norm in erp_by_name:
            erp_s = erp_by_name[plm_norm]
            match = {
                'canonical_ref': plm_norm,
                'source_system_a': plm_s['source_system'],
                'source_table_a': 'source_supplier',
                'raw_id_a': plm_s['source_reference'],
                'source_supplier_id_a': plm_s['id'],
                'source_system_b': erp_s['source_system'],
                'source_table_b': 'source_supplier',
                'raw_id_b': erp_s['source_reference'],
                'source_supplier_id_b': erp_s['id'],
                'relationship': 'identity',
                'confidence': 0.95,  # REAL
                'method': 'NORMALIZED',
                'status': 'PENDING',
            }
            matches.append(match)
            record_reconciliation(
                conn,
                entity_type='supplier',
                source_entity_id=plm_s['id'],
                status='PENDING',
                method='NORMALIZED',
                confidence=0.95,
                rationale=f"Identity with ERP source_supplier {erp_s['id']} on {plm_norm}",
                evidence=match,
            )

    return matches
```
