def detect_functional_similarity(conn: sqlite3.Connection, config: Optional[Dict] = None) -> List[Dict]:
    """Detect functionally similar but non-identical components.
    Queries source_component table per D-2.3.
    if config is None:
        config = load_normalization_config()

    candidates = []

    # Get all distinct normalized component references from source_component
    refs = conn.execute('''
        SELECT DISTINCT normalized_reference
        FROM source_component
        WHERE normalized_reference IS NOT NULL
        AND normalized_reference != ''
        ORDER BY normalized_reference
    ''').fetchall()

    ref_list = [r['normalized_reference'] for r in refs]

    # Find similar pairs: same prefix, different suffix
    for i, ref_a in enumerate(ref_list):
        for ref_b in ref_list[i+1:]:
            # Skip if one is alias of the other
            if ref_a == ref_b:
                continue

            # Check prefix similarity
            parts_a = ref_a.split('-')
            parts_b = ref_b.split('-')

            if len(parts_a) >= 2 and len(parts_b) >= 2:
                # Same prefix (first 2 parts), different suffix
                prefix_a = '-'.join(parts_a[:2])
                prefix_b = '-'.join(parts_b[:2])

                if prefix_a == prefix_b and parts_a[-1] != parts_b[-1]:
                    candidate = {
                        'reference_a': ref_a,
                        'reference_b': ref_b,
                        'similarity_type': 'same_prefix_different_suffix',
                        'similarity_detail': f"Prefix '{prefix_a}' matches, suffixes differ: '{parts_a[-1]}' vs '{parts_b[-1]}'",
                        'relationship': 'functional_similarity',
                        'confidence': 0.50,  # REAL 0–1
                        'method': 'STRUCTURED',  # canonical method enum (prefix heuristic)
                        'status': 'PENDING',  # PENDING | ASSESSED | ACCEPTED | REJECTED
                        'rationale': f"Similar component family but different variants — requires human review to determine if interchangeable",
                        'review_needed': True,  # evidence/app flag — not a DB column
                    }
                    candidates.append(candidate)

    return candidates
```

def detect_variant_specific_differences(conn: sqlite3.Connection) -> List[Dict]:
    """Detect components used only by specific variants (variant-specific specialization).

    Uses source_assembly_variant junction table to identify Nordic variant assemblies,
    then finds BOM lines exclusive to those variants.
    """
    # Get Nordic variant refs
    nordic_variants = conn.execute('''
        SELECT variant_ref_normalized
        FROM plm_variant
        WHERE variant_name_raw LIKE '%nordic%' OR variant_name_raw LIKE '%Nordic%'
        AND variant_ref_normalized IS NOT NULL
    ''').fetchall()

    nordic_refs = {v['variant_ref_normalized'] for v in nordic_variants}

    if not nordic_refs:
        return []

    # Find BOM lines used only by Nordic variants
    nordic_only_components = conn.execute('''
        SELECT DISTINCT component_ref_normalized
        FROM plm_bom_line
        WHERE variant_ref_normalized IN (SELECT variant_ref_normalized FROM plm_variant WHERE variant_name_raw LIKE '%nordic%' OR variant_name_raw LIKE '%Nordic%')
        AND component_ref_normalized NOT IN (
            SELECT component_ref_normalized
            FROM plm_bom_line
            WHERE variant_ref_normalized NOT IN (SELECT variant_ref_normalized FROM plm_variant WHERE variant_name_raw LIKE '%nordic%' OR variant_name_raw LIKE '%Nordic%')
        )
    ''').fetchall()

    records = []
    for comp in nordic_only_components:
        record = {
            'component_ref': comp['component_ref_normalized'],
            'relationship': 'variant_specific',
            'confidence': 0.95,  # REAL — was incorrectly TEXT 'high'
            'method': 'STRUCTURED',  # canonical method enum
            'status': 'PENDING',  # canonical — was incorrectly 'excluded'
            'variant_ref': list(nordic_refs)[0],
            'rationale': 'Component used exclusively by Nordic climate variant — intentional specialization, not a merge candidate',
            'review_needed': False,
        }
        records.append(record)

    return records
```
