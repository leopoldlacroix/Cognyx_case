```python
def detect_language(text: str) -> str:
    """Detect language of engineering note text.

    Heuristic-based: French (accents, common words), German (typical chars/words),
    default to English.
    """
    if not text:
        return 'EN'

    text_upper = text.upper()

    # French indicators: accented characters, common French words
    french_chars = set('ÉÈÊËÀÂÎÏÔÛÙÇ')
    french_words = {'LE', 'LA', 'LES', 'UN', 'UNE', 'DES', 'DU', 'DE', 'ET', 'OU',
                    'EST', 'SON', 'SES', 'DANS', 'SUR', 'POUR', 'AVEC', 'sans',
                    'NOTRE', 'NOTREMENT', 'CE', 'CETTE', 'CEST', 'IL', 'ELLE',
                    'VOUS', 'NOUS', 'EUROPE', 'EUROPÉEN'}

    # German indicators
    german_words = {'DER', 'DIE', 'DAS', 'EIN', 'EINE', 'UND', 'ODER', 'MIT',
                    'FÜR', 'AUF', 'IN', 'IST', 'SIND', 'HABEN', 'KÖNNEN',
                    'MÜSSEN', 'SICH', 'AUS', 'AN', 'AB', 'NOCH', 'SCHON',
                    'AACHEN', 'DEUTSCHLAND', 'GERMAN'}

    has_french_char = any(c in french_chars for c in text)
    text_words = set(text_upper.split())

    french_score = len(text_words & french_words)
    german_score = len(text_words & german_words)

    if has_french_char and french_score >= german_score:
        return 'FR'
    elif german_score > french_score and german_score >= 2:
        return 'DE'
    else:
        return 'EN'


def normalize_engineering_notes(conn: sqlite3.Connection) -> Dict[str, Any]:
    """Normalize engineering notes: detect language, clean text.

    For each engineering_note row:
    - Detect language from note_text
    - Populate language_normalized
    - Normalize text (whitespace collapse, punctuation cleanup)
    - Populate note_text_normalized
    """
    stats = {'processed': 0, 'languages': {}}

    for row in conn.execute('SELECT id, note_text FROM engineering_note WHERE note_text IS NOT NULL').fetchall():
        stats['processed'] += 1
        text = row['note_text']

        # Detect language
        lang = detect_language(text)
        conn.execute('UPDATE engineering_note SET language_normalized = ? WHERE id = ?',
                    (lang, row['id']))
        stats['languages'][lang] = stats['languages'].get(lang, 0) + 1

        # Normalize text: whitespace collapse, basic cleanup
        normalized_text = normalize_whitespace(text)
        conn.execute('UPDATE engineering_note SET note_text_normalized = ? WHERE id = ?',
                    (normalized_text, row['id']))

    conn.commit()
    return stats
```
