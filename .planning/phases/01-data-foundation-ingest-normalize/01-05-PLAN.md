import json
from pathlib import Path
from typing import Optional, Dict

def load_normalization_config() -> Dict:
    """Load normalization config from config/normalization.json."""
    config_path = Path(__file__).parent.parent / 'config' / 'normalization.json'
    with open(config_path) as f:
        return json.load(f)

def apply_uom_aliases(normalized_uom: Optional[str], config: Dict) -> Optional[str]:
    """Apply UOM aliases. Returns canonical UOM or original if no alias found."""
    if not normalized_uom:
        return normalized_uom
    
    uom_aliases = config.get('uom_aliases', {})
    return uom_aliases.get(normalized_uom, normalized_uom)

def apply_supplier_aliases(normalized_supplier: Optional[str], config: Dict) -> Optional[str]:
    """Apply supplier aliases with prefix matching.
    
    Returns canonical supplier name or original if no alias matches.
    Tries exact match first, then prefix match.
    """
    if not normalized_supplier:
        return normalized_supplier
    
    supplier_aliases = config.get('supplier_aliases', {})
    
    # Try exact match first
    if normalized_supplier in supplier_aliases:
        return supplier_aliases[normalized_supplier]
    
    # Try prefix matching (e.g., "SIEMENS MOBILITY GMBH" -> "SIEMENS MOBILITY")
    supplier_upper = normalized_supplier.upper()
    for alias, canonical in supplier_aliases.items():
        if supplier_upper.startswith(alias.upper()):
            return canonical
    
    return normalized_supplier

def apply_reference_aliases(normalized_reference: Optional[str], config: Dict) -> Optional[str]:
    """Apply reference aliases. Returns canonical reference or original if no alias found."""
    if not normalized_reference:
        return normalized_reference
    
    reference_aliases = config.get('reference_aliases', {})
    return reference_aliases.get(normalized_reference, normalized_reference)
```

NORMALIZATION_CONFIG = load_normalization_config()

def normalize_bom_row(row: dict, source_system: str) -> dict:
    """Apply normalization + aliases to a BOM row."""
    normalized = {}
    
    # Reference fields: generic normalize -> alias lookup
    raw_component_ref = row.get('component_ref_raw')
    generic_normalized = normalize_reference(raw_component_ref)
    normalized['component_ref_normalized'] = apply_reference_aliases(generic_normalized, NORMALIZATION_CONFIG)
    
    raw_variant_ref = row.get('variant_ref_raw')
    normalized['variant_ref_normalized'] = normalize_reference(raw_variant_ref)
    # Variant references don't have aliases per D-11
    
    raw_assembly_ref = row.get('assembly_ref_raw')
    normalized['assembly_ref_normalized'] = normalize_reference(raw_assembly_ref)
    
    # Description: generic normalize only (no aliases for descriptions)
    normalized['description_normalized'] = normalize_description(row.get('description_raw'))
    
    # UOM: generic normalize -> UOM alias lookup
    raw_uom = row.get('uom_raw')
    generic_uom = normalize_uom(raw_uom)
    normalized['uom_normalized'] = apply_uom_aliases(generic_uom, NORMALIZATION_CONFIG)
    
    # Supplier: generic normalize -> supplier alias lookup
    raw_supplier = row.get('supplier_raw')
    generic_supplier = normalize_supplier(raw_supplier)
    normalized['supplier_normalized'] = apply_supplier_aliases(generic_supplier, NORMALIZATION_CONFIG)
    
    # Quantity: parse as float
    qty_raw = row.get('quantity_raw')
    if qty_raw:
        try:
            normalized['quantity_normalized'] = float(str(qty_raw).replace(',', ''))
        except (ValueError, TypeError):
            normalized['quantity_normalized'] = None
    else:
        normalized['quantity_normalized'] = None
    
    return normalized
```
