import re
from typing import Optional

def normalize_whitespace(s: Optional[str]) -> Optional[str]:
    """Collapse multiple whitespace to single space, trim."""
    if not s:
        return s
    return re.sub(r'\s+', ' ', s).strip()

def normalize_case_identifier(s: Optional[str]) -> Optional[str]:
    """Uppercase for reference identifiers."""
    if not s:
        return s
    return normalize_whitespace(s).upper()

def normalize_case_description(s: Optional[str]) -> Optional[str]:
    """Preserve case but trim for descriptions."""
    if not s:
        return s
    return normalize_whitespace(s)

def normalize_punctuation(s: Optional[str]) -> Optional[str]:
    """Harmonize dashes, underscores, periods in identifiers."""
    if not s:
        return s
    # Normalize various dash/underscore forms to single dash
    s = re.sub(r'[-_]+', '-', s)
    # Remove leading/trailing punctuation
    s = s.strip('-.')
    return s

def normalize_reference(raw: Optional[str]) -> Optional[str]:
    """Apply generic normalization to reference identifiers.
    
    Order: whitespace -> case (upper) -> punctuation
    """
    if not raw:
        return raw
    s = normalize_whitespace(raw)
    s = s.upper()
    s = normalize_punctuation(s)
    return s

def normalize_description(raw: Optional[str]) -> Optional[str]:
    """Apply generic normalization to descriptions.
    
    Order: whitespace -> preserve case -> light punctuation cleanup
    """
    if not raw:
        return raw
    s = normalize_whitespace(raw)
    # For descriptions, just trim and collapse whitespace, preserve case
    return s

def normalize_uom(raw: Optional[str]) -> Optional[str]:
    """Normalize UOM — generic cleanup only (aliases in PLAN-1.5)."""
    if not raw:
        return raw
    return normalize_whitespace(raw).upper()

def normalize_supplier(raw: Optional[str]) -> Optional[str]:
    """Normalize supplier name — generic cleanup only (aliases in PLAN-1.5)."""
    if not raw:
        return raw
    return normalize_whitespace(raw)
```

from app.services.normalization import (
    normalize_reference,
    normalize_description,
    normalize_uom,
    normalize_supplier
)

def normalize_bom_row(row: dict, source_system: str) -> dict:
    """Apply normalization to a BOM row, returning normalized values."""
    normalized = {}
    
    # Reference fields — use normalize_reference (uppercase, punctuation harmonization)
    normalized['variant_ref_normalized'] = normalize_reference(row.get('variant_ref_raw'))
    normalized['assembly_ref_normalized'] = normalize_reference(row.get('assembly_ref_raw'))
    normalized['component_ref_normalized'] = normalize_reference(row.get('component_ref_raw'))
    
    # Description — preserve case, just whitespace cleanup
    normalized['description_normalized'] = normalize_description(row.get('description_raw'))
    
    # UOM — uppercase, whitespace cleanup
    normalized['uom_normalized'] = normalize_uom(row.get('uom_raw'))
    
    # Supplier — whitespace cleanup, preserve case
    normalized['supplier_normalized'] = normalize_supplier(row.get('supplier_raw'))
    
    # Quantity — parse as float when possible
    qty_raw = row.get('quantity_raw')
    if qty_raw:
        try:
            normalized['quantity_normalized'] = float(str(qty_raw).replace(',', ''))
        except (ValueError, TypeError):
            normalized['quantity_normalized'] = None
    else:
        normalized['quantity_normalized'] = None
    
    return normalized
```

In the BOM line ingestion loop:
1. After reading raw row, call normalize_bom_row()
2. Insert raw values into *_raw columns
3. Insert normalized values into *_normalized columns

class TestNormalizationIntegration:
    def test_normalized_values_in_database(self, fresh_db, sample_data_paths):
        from app.services.ingestion import ingest_csv
        
        # Ingest BOM export
        ingest_csv(sample_data_paths['bom_export'], 'PLM', fresh_db)
        
        # Query raw and normalized values
        rows = fresh_db.execute('''
            SELECT component_ref_raw, component_ref_normalized, 
                   description_raw, description_normalized
            FROM plm_bom_line
            LIMIT 5
        ''').fetchall()
        
        assert len(rows) > 0, "Expected BOM lines in database"
        
        for row in rows:
            # Raw value should exist
            assert row['component_ref_raw'] is not None
            # Normalized value should exist for non-null raw
            if row['component_ref_raw']:
                assert row['component_ref_normalized'] is not None
                # Normalized should be uppercase
                assert row['component_ref_normalized'] == row['component_ref_normalized'].upper()
    
    def test_raw_values_preserved(self, fresh_db, sample_data_paths):
        from app.services.ingestion import ingest_csv

            SELECT component_ref_raw, description_raw, uom_raw
            FROM plm_bom_line
            WHERE component_ref_raw = 'CTRL-AIR-01'
        ''').fetchall()
        
        assert len(rows) > 0, "Expected CTRL-AIR-01 in database"
        
        for row in rows:
            # Raw should be exactly as in CSV
