  4. Structurally malformed row: CSV row that cannot be parsed at all

- Rows with empty component_ref_raw ARE hard-quarantined
- The function does NOT modify the row — it only returns a reason or None
</acceptance_criteria>

<action>
Create `app/services/validation.py` with:

```python
from typing import Optional, Dict, Any

def validate_hard(row: Dict[str, Any], source_system: str, table_name: str) -> Optional[str]:
    """
    Run hard validation checks on a parsed row.
    Returns rejection reason string if row fails, None if passes.
    """
    # Rule 1: Missing required identifier
    required_id_fields = {
        'plm_bom_line': ['component_ref_raw'],
        'plm_assembly': ['assembly_ref_raw'],
        'plm_variant': ['variant_ref_raw'],
        'erp_material': ['material_id_raw'],
        'erp_supplier': ['supplier_id_raw'],
        'engineering_note': ['object_reference_raw'],
    }
    
    if table_name in required_id_fields:
        for field in required_id_fields[table_name]:
            value = row.get(field, '')
            if value is None or str(value).strip() == '':
                return f"Missing required identifier: {field}"
    
    # Rule 2: Unparseable quantity (BOM lines only)
    if table_name == 'plm_bom_line':
        qty_raw = row.get('quantity_raw', '')
        if qty_raw is not None and str(qty_raw).strip() != '':
            try:
                float(str(qty_raw).strip())
            except ValueError:
                return f"Unparseable quantity: '{qty_raw}'"
    
    return None
```

```python
def get_quarantine_report(conn, source_file_id=None):
    """Return quarantine records with file provenance."""
    if source_file_id:
        rows = conn.execute("""
            SELECT q.id, q.source_file_id, q.source_row, q.rejection_reason,
                   q.created_at, sf.file_name, sf.source_system
            FROM quarantine q
            JOIN source_file sf ON q.source_file_id = sf.id
            WHERE q.source_file_id = ?
            ORDER BY q.source_row
        """, (source_file_id,)).fetchall()
    else:
        rows = conn.execute("""
            SELECT q.id, q.source_file_id, q.source_row, q.rejection_reason,
                   q.created_at, sf.file_name, sf.source_system
            FROM quarantine q
            JOIN source_file sf ON q.source_file_id = sf.id
            ORDER BY sf.file_name, q.source_row
        """).fetchall()
    return [dict(r) for r in rows]

def get_quarantine_count(conn, source_file_id=None):
    if source_file_id:
        return conn.execute(
            'SELECT COUNT(*) as c FROM quarantine WHERE source_file_id = ?',
            (source_file_id,)
        ).fetchone()['c']
    return conn.execute('SELECT COUNT(*) as c FROM quarantine').fetchone()['c']
```
